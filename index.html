import { Router } from 'express'; import { createInvoice, getInvoices, recordPayment } from '../controllers/invoiceController'; const router = Router(); // Routes Definition router.post('/invoices', createInvoice); // POST /api/v1/invoices router.get('/invoices', getInvoices); // GET /api/v1/invoices?status=PENDING router.post('/invoices/pay', recordPayment); // POST /api/v1/invoices/pay export default router;
