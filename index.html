import express from 'express'; import cors from 'cors'; import invoiceRoutes from './routes/invoiceRoutes'; const app = express(); const PORT = process.env.PORT || 5000; // Middlewares app.use(cors()); app.use(express.json()); // API Routes app.use('/api/v1', invoiceRoutes); app.get('/health', (req, res) => { res.status(200).json({ status: 'OK', timestamp: new Date() }); }); app.listen(PORT, () => { console.log(`ERP Backend API running on port ${PORT}`); });
